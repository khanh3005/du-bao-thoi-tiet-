@st.cache_resource(show_spinner="Đang tải mô hình phân loại...")
def load_artifacts():
	if not MODEL_PATH.is_file():
		raise FileNotFoundError(f"Không tìm thấy mô hình: {MODEL_PATH.name}")
	if not CLASSES_PATH.is_file():
		raise FileNotFoundError(f"Không tìm thấy danh sách nhãn: {CLASSES_PATH.name}")
	model = joblib.load(MODEL_PATH)
	classes = np.load(CLASSES_PATH, allow_pickle=True).astype(str)
	return model, classes


def preprocess_image(image_bytes: bytes) -> np.ndarray:
	encoded = np.frombuffer(image_bytes, dtype=np.uint8)
	image_bgr = cv2.imdecode(encoded, cv2.IMREAD_COLOR)
	if image_bgr is None:
		raise ValueError("Không thể đọc ảnh này. Hãy thử tệp PNG hoặc JPEG hợp lệ.")
	image_rgb = cv2.cvtColor(cv2.resize(image_bgr, IMAGE_SIZE), cv2.COLOR_BGR2RGB)
	return (image_rgb / 255.0).reshape(1, -1)


def label_name(encoded_label, classes: np.ndarray) -> str:
	try:
		index = int(encoded_label)
		if 0 <= index < len(classes):
			return str(classes[index])
	except (TypeError, ValueError):
		pass
	return str(encoded_label)

			features = preprocess_image(uploaded_file.getvalue())
			prediction = model.predict(features)[0]
			predicted_label = label_name(prediction, classes)

			if hasattr(model, "predict_proba"):
				probabilities = model.predict_proba(features)[0]
				model_classes = getattr(model, "classes_", range(len(probabilities)))
				ranked = sorted(
					((label_name(class_id, classes), float(score)) for class_id, score in zip(model_classes, probabilities)),
					key=lambda item: item[1],
					reverse=True,
				)
